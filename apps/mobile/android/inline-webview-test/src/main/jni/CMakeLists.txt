cmake_minimum_required(VERSION 3.22)
project(appmodules)
find_package(ReactAndroid REQUIRED CONFIG)
find_package(fbjni REQUIRED CONFIG)
add_library(appmodules SHARED OnLoad.cpp)
target_compile_features(appmodules PRIVATE cxx_std_20)
target_compile_options(appmodules PRIVATE -fexceptions -frtti)
target_compile_definitions(appmodules PRIVATE FOLLY_NO_CONFIG=1 FOLLY_HAVE_CLOCK_GETTIME=1)
target_link_libraries(appmodules ReactAndroid::reactnative ReactAndroid::jsi fbjni::fbjni)
